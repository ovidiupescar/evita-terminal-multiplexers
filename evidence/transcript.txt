===== PART 1 - WHY A MULTIPLEXER: A COMMAND THAT OUTLIVES ITS TERMINAL =====

--- the fixture: a command that keeps writing until it is stopped ---
$ cat ticker.sh
#!/bin/bash
# prints one line per second until it is stopped
i=0
while true; do
  i=$((i+1))
  printf "%s tick %d\n" "$(date +%H:%M:%S)" "$i"
  sleep 1
done
[exit 0]
$ ls -l ticker.sh
-rwxr-xr-x 1 ovidiu ovidiu 157 Jul 31 10:04 ticker.sh
[exit 0]

--- CASE A: the command runs on a pseudo-terminal that is then destroyed.
--- script(1) holds the pseudo-terminal, which stands in for the terminal
--- an SSH connection provides. Destroying it delivers the same SIGHUP a
--- dropped connection delivers.
$ setsid script -q -c "$HOME/use17/ticker.sh > $HOME/use17/plain.log 2>&1" /dev/null < /dev/null > /dev/null 2>&1 & echo "harness pid $!" > "$HOME/.use17-apid"; sleep 4; cat "$HOME/.use17-apid"
harness pid 312
[exit 0]
$ pstree -p 312
script(312)---bash(314)---ticker.sh(315)---sleep(369)
[exit 0]
$ cat plain.log; echo "-- $(wc -l < plain.log) lines"
10:04:05 tick 1
10:04:06 tick 2
10:04:07 tick 3
10:04:08 tick 4
-- 4 lines
[exit 0]
$ kill 312
[exit 0]
$ sleep 3; cat plain.log; echo "-- $(wc -l < plain.log) lines"
10:04:05 tick 1
10:04:06 tick 2
10:04:07 tick 3
10:04:08 tick 4
10:04:09 tick 5
-- 5 lines
[exit 0]
$ pgrep -af ticker.sh || echo "no ticker.sh process is running"
no ticker.sh process is running
[exit 0]
