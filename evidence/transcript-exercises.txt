===== EX3 - finish an edit across a disconnection =====

[exit 0]
$ tmux new-session -d -s remote -x 80 -y 24 -c "$HOME/use17" "nano settings.conf"
[exit 0]
$ sleep 2; tmux send-keys -t remote Down End Enter; sleep 1; tmux send-keys -t remote -l "threads = 4"; sleep 1; tmux ls
